import concurrent.futures,subprocess,pathlib,hashlib,json,time,os
root=pathlib.Path('/extra/scylladb/gocql')
changed=subprocess.check_output(['git','diff','--name-only','HEAD'],cwd=root,text=True).splitlines()
added=subprocess.check_output(['git','ls-files','--others','--exclude-standard'],cwd=root,text=True).splitlines()
paths=sorted(set(p for p in changed+added if p.endswith('.go') or p in ['feature-specs/gocql-v2-api.md','feature-specs/gocql-v2-implementation.md']))
snapshot={p:hashlib.sha256((root/p).read_bytes()).hexdigest() for p in paths if (root/p).is_file()}
(root/'feature-specs/evidence/step04-release-source.json').write_text(json.dumps(snapshot,indent=2)+'\n')

results=[]
def run(item):
 name,cmd,env=item;log=root/f'feature-specs/evidence/step04-release-{name}.log';start=time.time()
 with log.open('w') as out:p=subprocess.run(cmd,cwd=root,env=dict(os.environ,**env),stdout=out,stderr=subprocess.STDOUT)
 return {'name':name,'command':cmd,'environment':env,'exit_code':p.returncode,'elapsed_seconds':time.time()-start,'log':str(log.relative_to(root)),'source_snapshot':'feature-specs/evidence/step04-release-source.json'}
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
 for result in pool.map(run,checks):
  results.append(result);(root/'feature-specs/evidence/step04-release-check-results.json').write_text(json.dumps(results,indent=2)+'\n');print(json.dumps(result),flush=True)
