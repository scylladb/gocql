import subprocess,hashlib,json,pathlib,time
root=pathlib.Path('/extra/scylladb/gocql')
base=['go','test','-json','-race','-tags','integration gocql_debug','-run','^(TestV2TransportPoolSizingLive|TestV2ExclusiveTransportLive)$','-count=1','-timeout=8m','-parallel=1','.','-args','-distribution=scylla','-gocql.timeout=60s','-proto=4','-rf=3','-clusterSize=3','-autowait=2000ms','-compressor=no-compression','-gocql.cversion=2026.3.4','-cluster=127.0.20.1,127.0.20.2,127.0.20.3']
paths=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard','*.go'],cwd=root,text=True).splitlines()
snapshot={p:hashlib.sha256((root/p).read_bytes()).hexdigest() for p in paths if (root/p).is_file()}
(root/'feature-specs/evidence/step04-live-initial-source.json').write_text(json.dumps(snapshot,indent=2)+'\n')
log=root/'feature-specs/evidence/step04-live-stable-raw-initial.jsonl'
start=time.time()
with log.open('w') as out: result=subprocess.run(base,cwd=root,stdout=out,stderr=subprocess.STDOUT)
record={'command':base,'exit_code':result.returncode,'log':str(log.relative_to(root)),'elapsed_seconds':time.time()-start,'source_snapshot':'feature-specs/evidence/step04-live-initial-source.json'}
(root/'feature-specs/evidence/step04-live-initial-results.json').write_text(json.dumps([record],indent=2)+'\n')
print(json.dumps(record))
